```yaml

  modelo: SaaS
  arquitetura: multiempresa

principais_recursos:
  - gerenciamento de empresas
  - gerenciamento de unidades
  - playlists remotas
  - biblioteca musical
  - gravações
  - anúncios
  - programação
  - player por unidade
  - sincronização remota
  - telemetria
  - usuários e permissões
```
